bool checksumOK(const String &frame) {
  // Frame format:
  // 02 [ASCII payload] [CHK] [INVCHK] 03

  int len = frame.length();

#include <SPI.h>
#include <SD.h>

const int SD_CS = SDCARD_SS_PIN;
const int WL134_RESET_PIN = 6;

// ======================================================
// TAG EXTRACTION (your working method)
// ======================================================
String extractTagID(const String &frame) {

  int start = 1;  // after STX (0x02)
  int end = frame.lastIndexOf('\x03');  // ETX (0x03)
  String ascii = frame.substring(start, end);

  // First 13 characters only
  String raw = ascii.substring(0, 13);

  // Reverse character-by-character
  String rev = "";
  for (int i = raw.length() - 1; i >= 0; i--) {
    rev += raw[i];
  }

  // Insert period after first 3 characters
  String tag = rev.substring(0, 3) + "." + rev.substring(3);

  return tag;
}

// ======================================================
// CHECKSUM VALIDATION
// ======================================================
bool checksumOK(const String &frame) {
  int len = frame.length();
  if (len < 5) return false;

  uint8_t chk = frame[len - 3];
  uint8_t inv = frame[len - 2];

  return ((chk ^ inv) == 0xFF);
}

// ======================================================
// TIMESTAMP (placeholder — can replace with RTC later)
// ======================================================
String timestamp() {
  unsigned long ms = millis();
  char buf[32];
  sprintf(buf, "%lu", ms);
  return String(buf);
}

// ======================================================
// CSV LOGGING
// ======================================================
void logCSV(const String &ts, const String &tag) {
  File f = SD.open("rfidlog.csv", FILE_WRITE);
  if (f) {
    f.print(ts);
    f.print(",");
    f.println(tag);
    f.close();
  }
}

// ======================================================
// RESET PULSE
// ======================================================
void pulseReset() {
  digitalWrite(WL134_RESET_PIN, LOW);
  delay(40);
  digitalWrite(WL134_RESET_PIN, HIGH);
}

// ======================================================
// SETUP
// ======================================================
void setup() {
  Serial.begin(115200);
  Serial1.begin(9600);

  pinMode(WL134_RESET_PIN, OUTPUT);
  digitalWrite(WL134_RESET_PIN, HIGH);

  if (!SD.begin(SD_CS)) {
    Serial.println("SD Card init failed!");
  } else {
    Serial.println("SD Card ready.");
  }

  Serial.println("WL-134 PIT Tag Logger Ready.");
}

// ======================================================
// MAIN LOOP — reset once per second, read tag
// ======================================================
unsigned long lastReset = 0;

void loop() {

  // Reset WL-134 once per second
  if (millis() - lastReset >= 1000) {
    pulseReset();
    lastReset = millis();
  }

  // Read frame if available
  if (Serial1.available()) {

    String frame = "";
    while (Serial1.available()) {
      char c = Serial1.read();
      frame += c;
      if (c == '\x03') break;  // ETX
    }

    if (frame.length() < 10) return;
    if (!checksumOK(frame)) return;

    // Extract tag
    String tag = extractTagID(frame);
    String ts = timestamp();

    // Print to Serial
    Serial.print(ts);
    Serial.print(" TAG: ");
    Serial.println(tag);

    // Log to SD (CSV)
    logCSV(ts, tag);
  }
}
